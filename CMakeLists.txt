cmake_minimum_required(VERSION 3.16)
project(ollama_xp C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# third_party/llama.cpp is fetched by scripts/fetch-llama.sh. Without it the program still builds
# with a mock engine (useful to test the CLI / REST API, but it cannot run real models).
option(XPLLM_WITH_LLAMA "Link llama.cpp" ON)
set(LLAMA_DIR ${CMAKE_SOURCE_DIR}/third_party/llama.cpp)

set(SRC src/main.cpp src/json.cpp src/util.cpp src/store.cpp src/template.cpp src/engine.cpp
        src/engine_mock.cpp src/engine_llama.cpp src/runner.cpp src/http.cpp src/pull.cpp src/server.cpp src/xp_compat.c)
add_executable(ollama ${SRC})

if(XPLLM_WITH_LLAMA AND EXISTS ${LLAMA_DIR}/CMakeLists.txt)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
  # Old CPUs: no AVX family by default. Enable with -DXPLLM_SSE2=ON for Pentium 4 / Athlon 64 and newer.
  option(XPLLM_SSE2 "Compile with SSE2" ON)
  foreach(f AVX AVX2 AVX512 FMA F16C BMI2)
    set(GGML_${f} OFF CACHE BOOL "" FORCE)
  endforeach()
  if(XPLLM_SSE2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "i[3-6]86")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -msse2 -mfpmath=sse")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -msse2 -mfpmath=sse")
  endif()
  add_subdirectory(${LLAMA_DIR} llama-build EXCLUDE_FROM_ALL)
  target_include_directories(ollama PRIVATE ${LLAMA_DIR}/include ${LLAMA_DIR}/ggml/include)
  target_link_libraries(ollama PRIVATE llama)
  target_compile_definitions(ollama PRIVATE XPLLM_HAVE_LLAMA=1)
  message(STATUS "llama.cpp engine: ON")
else()
  message(WARNING "llama.cpp not found in third_party/ (run scripts/fetch-llama.sh) - building with MOCK engine")
endif()

if(WIN32)
  target_link_libraries(ollama PRIVATE ws2_32)
  target_compile_definitions(ollama PRIVATE _WIN32_WINNT=0x0501 WINVER=0x0501)
endif()
find_package(Threads REQUIRED)
target_link_libraries(ollama PRIVATE Threads::Threads)

option(XPLLM_TESTS "Build unit tests" OFF)
if(XPLLM_TESTS)
  add_executable(test_core tests/test_core.cpp src/json.cpp src/util.cpp src/store.cpp src/template.cpp src/engine.cpp)
  target_link_libraries(test_core PRIVATE Threads::Threads)
endif()
